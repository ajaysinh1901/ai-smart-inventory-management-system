A React Context Provider (State Management)
AuthContext.jsx
  1. import { createContext, useContext, useState } from 'react';
  3. const AuthContext = createContext();
  5. export function AuthProvider({ children }) {
  6. const [user, setUser] = useState(null);
  8. const login = (data) => setUser(data);
  9. const logout = () => setUser(null);
  11. return (
  12. <AuthContext.Provider value={{ user, login, logout }}>
  13. {children}
  14. </AuthContext.Provider>
  15. );
  16. }
  18. export const useAuth = () => useContext(AuthContext);
