A Simple Node.js Web Server
server.js
  1. const http = require('http');
  3. // A minimal web server using the built-in http module
  4. const server = http.createServer((req, res) => {
  5. res.writeHead(200, { 'Content-Type': 'text/plain' });
  6. res.end('Hello from the Node.js server');
  7. });
  9. server.listen(5000, () => {
  10. console.log('Server running on port 5000');
  11. });
